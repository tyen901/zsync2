enable_testing()
include(CTest)

set(CMAKE_CXX_STANDARD 17)

find_package(GTest REQUIRED)

add_executable(test_zsutil test_zsutil.cpp)
if(WIN32)
	target_link_libraries(test_zsutil PRIVATE libzsync2_static GTest::gtest_main cpr)
else()
	target_link_libraries(test_zsutil PRIVATE libzsync2 GTest::gtest_main cpr)
endif()
gtest_discover_tests(test_zsutil)
